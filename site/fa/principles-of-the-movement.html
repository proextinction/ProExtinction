نخست رنج را تعریف کنیم تا هیچ‌کس نتواند با بازی با کلمات این ایدئولوژی را گنگ کند.

رنج در تعریف ما و در سراسر جنبش و وب‌سایت ما، هر احساس بدی است که یک موجود حس‌مند می‌خواهد از آن دوری کند. ما آن را از درد جدا می‌کنیم. برای نمونه، یک مازوخیست نمی‌خواهد از برخی دردها دوری کند، اما حتی او هم نمی‌خواهد رنج ببرد. پس هر احساس بدی که یک موجود آگاه می‌خواهد از آن دوری کند رنج است. و این می‌تواند هر چیزی باشد، از درد جسمی تا پریشانی ذهنی و عاطفی که موجودات حس‌مند می‌خواهند از آن دوری کنند.

بر پایهٔ تعریف بالا، رنج تنها چیز بد در این جهان است. چرا گرسنگی بد است؟ رنج. چرا بیماری‌ها بدند؟ چرا جنایت بد است؟ چرا تجاوز به کودک بد است؟ چرا آزار حیوانات بد است؟ چرا شکار بد است؟ چرا بلایای طبیعی بدند؟ به خاطر رنج.

پس اگر رنج تنها چیز بد است، اخلاق رنج‌محور تنها اخلاق منطقی است. به‌ویژه وقتی چیزهای خوب چیزهای بد را توجیه نمی‌کنند. چیزهای خوب این جهان هرگز نمی‌توانند چیزهای بد را توجیه کنند، همان‌طور که سکس نمی‌تواند تجاوز به کودک را توجیه کند. غذای خوشمزه نمی‌تواند شکار یا بردگی حیوانات را توجیه کند. بازی‌های ویدیویی نمی‌توانند اختلال‌های روانی را توجیه کنند.

پس تنها چیز بد در این جهان رنج یا رنج احتمالی است.

و اخلاق رنج‌محور، برخلاف بسیاری از اخلاق‌های دیگر، بر نگرش "همینی است که هست" که همهٔ اخلاق‌های دیگر دنبال می‌کنند بنا نشده است. برای نمونه: اخلاق دینی می‌گوید "همجنس‌گرا بودن بد است چون خدا چنین گفته"، یا اخلاق پاکدامنی مانند "تا ازدواج سکس نکن، گناه است"، یا اخلاق میهن‌پرستانه "دوست نداشتن کشورت بد است"، این‌ها هیچ توجیهی ندارند. اما SFE را علم و ماده‌گرایی پشتیبانی می‌کنند. این‌که موجودات حس‌مند می‌خواهند از رنج دوری کنند یک واقعیت مادی است، یعنی واقعی است، نه خیالی مانند دین یا ملت‌ها. موجودات حس‌مند واقعی‌اند. آن‌ها رنج می‌برند. یک کتاب دینی یا پرچم کشور یا طبیعت یا باکرگی یک فرد رنج نمی‌برد! اما موجودات زندهٔ حس‌مند رنج می‌برند! و این آن‌ها را تنها واقعیت‌های ارزشمند این جهان می‌کند. هیچ چیز ارزشمندتر از یک موجود حس‌مند نیست.

پس به‌طور خلاصه:

  • رنج بد است
  • رنج تنها چیز بد است
  • SFE تنها اخلاق منطقی و مبتنی بر واقعیت است
  • موجودات حس‌مند از هر شیء غیرحس‌مندی ارزشمندترند.

ماده‌گرایی

این آشکارترین و در عین حال نادیده‌گرفته‌ترین منطق است. ماده‌گرایی ساده است. یعنی جادو وجود ندارد. این بنیاد خود علم است، یعنی مطالعهٔ واقعیت مادی ما.

ماده‌گرایی می‌گوید جهان پیرامون ما از قانون‌های فیزیکی ساخته شده و هیچ چیز فراطبیعی‌ای در آن نیست. انسان بود که مفهوم‌های فراطبیعی را ساخت، اما علم دارد آن‌ها را می‌شکند. باور داشتیم روح داریم. اما اکنون علم دریافته که ما کجا ساکنیم: مغز.

پژوهش علمی واقعی پرتلاش و عقلانی است. فرض نمی‌کند و جاهای خالی را پر نمی‌کند. زمان واقعی صرف پژوهش و یافتن حقیقت می‌کند، یعنی قانون‌های فیزیکی‌ای که بر ما حاکم‌اند. دین، معنویت و مانند آن‌ها تنبل و بی‌صبرند، فقط جاهای خالی را با هر داستان جادویی که پیدا کنند و بی هیچ دلیلی پر می‌کنند.

چرا ماده‌گرایی برای انقراض‌گرایی مهم است؟

چون جنبش انقراض‌گرا همیشه باید عقلانی بماند. ایدئولوژی‌های مزخرف زیادی هست، مانند زندگی پس از مرگ، کارما، اختیار آزاد، خداباوری، معنویت و مانند آن‌ها، که افرادی کاملاً غیرعقلانی دارند. آن‌ها برای جنبشی مبتنی بر عقلانیت و علم بی‌فایده‌اند. پس اگر کسی می‌خواهد به جنبش کمک کند، باید عقلانی و علم‌محور باشد.

جادو وجود ندارد. دیروز خورشیدگرفتگی خشم خدا بود که قربانی می‌خواست. امروز ماه است که میان خورشید و زمین قرار می‌گیرد.

هر چیزی ویژگی فیزیکی کشف‌شده یا کشف‌نشده‌ای دارد و بر پایهٔ آن ویژگی‌ها و قانون‌ها رفتار می‌کند. فراتر از واقعیت‌ها و قانون‌های فیزیکی چیزی نیست. جادو از نظر فیزیکی ناممکن است.

کوتاه همهٔ ایده‌های فراطبیعی غیرماده‌گرایانه را رد می‌کنم.

زندگی پس از مرگ/تناسخ

علم ثابت کرده که اطلاعات ما، یعنی "من"، در مغز ما جای دارد. پس از مرگ آن، هیچ چیز نیست. مغزتان در گل می‌پوسد. درست مانند پاک کردن همیشگی یک فایل. دیگر هیچ چیز نیست. اطلاعات می‌تواند نابود شود. زندگی پس از مرگ یا تناسخ سازوکاری برای کنار آمدن است که از ناتوانی در تصور نیستی یا ترس از نیستی برمی‌خیزد.

کارما

سازوکار دیگری برای کنار آمدن که از ناتوانی در مجازات آدم‌های بد برمی‌خیزد. نمی‌توانید کاری با سادیستی بکنید که حیوانات را شکنجه می‌کند، پس برایش بدبختی آرزو می‌کنید. اما در تاریخ آدم‌های زیادی بوده‌اند که مرگ‌های بسیاری به بار آوردند، زندگی شادی داشتند و با مرگی نسبتاً بی‌درد مردند. برخی باور دارند در زندگی بعدی مجازات می‌شوید. اما نمی‌دانم نظامی که مردم را به کارهای وحشتناک وامی‌دارد، سپس حافظه‌شان را پاک می‌کند و مجازاتشان می‌کند چه فایده‌ای دارد. و بیشترین رنج حتی به دست مردم نیست، به دست طبیعت است.

معنویت

این در واقع ایدئولوژی نیست، چون از آدم‌های بسیار تعریف‌های بسیاری دارد. این در قرن بیست‌ویکم کسب‌وکاری چند میلیارد دلاری است. کسانی که مشکلات روانی و ناامنی دارند به فرقه‌های معنوی کشیده می‌شوند که وعدهٔ "رهایی ذهن" می‌دهند. اما درمان‌های واقعی و ماده‌گرایانهٔ مشکلات روانی در داروهای روان‌پزشکی و روان‌درمانی است. هیچ جادویی نیست که بتواند "ذهنتان را رها کند"، اختلال‌ها از عدم تعادل در مغز ناشی می‌شوند.

حس‌مندمحوری

حس‌مندمحوری یعنی باید همهٔ موجودات حس‌مند را بی‌تبعیض در ملاحظهٔ اخلاقی بگنجانیم. صرف‌نظر از نژاد، جنسیت، ظاهر، گونه و مانند آن‌ها، هیچ‌کس هرگز نباید رنج ببرد. پس چیزهای غیرحس‌مند مانند یک سنگ یا انجیل از نظر اخلاقی بی‌اهمیت‌اند. نه یک حشره، نه یک پرنده، نه یک سگ، نه یک خوک، نه یک ماهی، نه موجودات حس‌مند فرازمینی احتمالی و نه هیچ موجود حس‌مند دیگری نباید به هستی و رنج شدید وادار شود.

بنابراین باید انقراض همهٔ حیات را به شیوه‌ای بی‌تبعیض دنبال کنیم، چون عواملی مانند هوش، ظاهر یا گونه در برابر آسیب‌پذیری ناچیزند و فقط حس‌مندی اهمیت دارد.،

جبرگرایی عصب‌زیستی

مغز ما یک شیء فیزیکی است. تفاوت‌های فیزیکی در مغز ما تفاوت‌های ذهنی را می‌سازند. برخی در ریاضی خوب‌اند، برخی نه. چرا این تفاوت پیش می‌آید؟ به خاطر تفاوت‌های مغزشان. کسی که در ریاضی خوب است ژن‌های خاصی را به‌طور فیزیکی دارد و بخش‌هایی از مغزش بهتر از کسانی که در ریاضی بدند رشد می‌کند. این تفاوت‌های فیزیکی همان چیزهایی‌اند که تفاوت در شخصیت، رفتار و مانند آن را می‌سازند. مغز فیزیکی سگ توانایی‌های کمتری از انسان دارد، پس انسان‌ها از سگ‌ها باهوش‌ترند. افرادی که ناتوانی ذهنی دارند نقص‌های فیزیکی در مغزشان دارند، همین است که ناتوانی ذهنی‌شان را پدید می‌آورد. کسانی که در هنر خوب یا بدند، این‌که تاب‌آوری بیشتری دارند، این‌که ارادهٔ بیشتری برای رفتن به باشگاه دارند، همه چیز را توانایی فیزیکی مغز تعیین می‌کند. توانایی‌ها یا ناتوانی‌های ذهنی از توانایی‌ها و ناتوانی‌های فیزیکی مغز برمی‌خیزند. پس از این‌جا به بعد، ناتوانی‌های ذهنی را ناتوانی‌های جسمی می‌نامم.

به همین شکل، حتی اخلاق، همدلی و عقلانیتی که کسی برای انقراض‌گرا بودن لازم دارد را هم همان مغز فیزیکی تعیین می‌کند. برای نمونه، یک سادیست سگی شکنجه‌شده را می‌بیند و می‌گوید "هی، این بد نیست"، چون در برخی عصب‌هایی که نورون‌های آینه‌ای را به آمیگدال وصل می‌کنند، که پاسخ عاطفی می‌سازد و همدلی پدید می‌آورد، نقص عصبی دارد. این تفاوت من با یک سادیست است. من شکنجهٔ توله‌سگ‌ها را بد می‌بینم. افزون بر این، او شاید اختلال سادیسم (باز هم اختلالی جسمی) داشته باشد که او را به شکنجهٔ توله‌سگ‌ها وامی‌دارد. از نظر نظری، اگر همان نقص‌ها در مغز من پدید آیند، من هم سادیست می‌شوم. چون مغز من فقط شیء دیگری است که فرایندی را همان‌طور که باید انجام می‌دهد. یک مدار برقی را عوض می‌کنید تا به جای بخاری، لامپ را روشن کند، مدار برقی دیگر اختیار آزاد ندارد که بخاری را روشن کند. درست همین‌طور، اگر مدارهای مغز من عوض شوند، دیگر اختیار آزاد ندارم که توله‌سگ را شکنجه نکنم. همان‌طور که فردی بی‌پا نمی‌تواند بدود، یک سادیست نمی‌تواند با سگی شکنجه‌شده همدلی کند.

این واقعیتی است که علم ثابت کرده است. پژوهش‌های بسیاری دربارهٔ نژادپرستی انجام شده است. برای نمونه، Phelps et al. 2000 و William Cunningham و همکاران 2004. این کارها دربارهٔ سوگیری و فعال‌سازی‌ای است که آمیگدال پس از دیدن عکس افرادی از نژادهای دیگر پدید می‌آورد. Richeson et al. 2003 قشر پیش‌پیشانی (PFC) را به عنوان ترمز بررسی می‌کند. پس وقتی فعال‌سازی آمیگدال اوج می‌گیرد، مردم نسبت به نژادهای دیگر پرخاشگر می‌شوند. سپس PFC باید آن را آرام کند. اگر نکند چه می‌شود؟ نسبت به نژادهای دیگر پرخاشگرانه رفتار خواهند کرد. اگر dlPFC با فعال‌سازی آمیگدال مقابله کند چه؟ آن افراد در درون هنوز نژادپرست خواهند بود، اما هوشمندانه خود را از پرخاشگری بازمی‌دارند. نقش mPFC چیست؟ افرادی با mPFC بهتر می‌توانند داوری اخلاقی کنند و فکر کنند "این‌ها فقط آدم‌هایی از نژادهای دیگرند، دلیلی برای پرخاشگری نیست" (Harris and Fiske 2006). می‌بینید رشد و چیرگی بخش‌های گوناگون مغز چگونه رفتار متفاوتی می‌سازد؟ پس اخلاق، همدلی و عقلانیت شما به معنای واقعی کلمه با توانایی فیزیکی مغزتان تعیین می‌شود. چیرگی بیشتر آمیگدال = پرخاشگری بیشتر. رشد بیشتر mPFC = داوری‌های اخلاقی خوب. این‌ها واقعیت‌های علمی انکارناپذیری‌اند که پژوهش ثابت کرده است.

پس چه چیزی این شخصیت‌ها و ویژگی‌های فیزیکی مغز را تعیین می‌کند؟

ژنتیک: ژن‌هایی که از پدر و مادرتان به ارث می‌برید. به همین دلیل پدیدهٔ دیدنی وراثت رخ می‌دهد: "مثل مادرش مهربان است"، "شبیه پدرش است". جدا از این، گاهی حتی می‌توانید ژن‌های جهش‌یافتهٔ خودتان را داشته باشید که ویژگی تازه یا ناتوانی‌ای پدید می‌آورند.

رشد: ورودی‌هایی که هنگام رشد مغز به آن می‌رسند. مانند این‌که تغذیهٔ خوبی داشته‌اید یا نه، کودکی بی‌آسیبی داشته‌اید یا نه، تعامل اجتماعی زیادی داشته‌اید یا نه و مانند آن.

پس دو انسان عاقل و اخلاق‌مدار که فرزندآوری می‌کنند و محیط خوبی مانند تغذیه، در دسترس بودن عاطفی و مانند آن به کودک می‌دهند، احتمال اخلاق‌مدار و عاقل بودن کودک را هم بیشتر می‌کنند.

پس این برای انقراض‌گرایی چه معنایی دارد؟

این حقیقت مطلق عصب‌زیست‌شناسی جبرگرا اثر زیادی بر اخلاق، راهبرد و علم جنبش دارد. و نیفزودن آن به اصول بی‌گمان در آینده بر تصمیم‌های جنبش اثر منفی می‌گذارد. کاربردهای کنونی:

اخلاق: جبرگرایی عصب‌زیستی مستقیماً یعنی یک سادیست سادیست می‌ماند، یک متجاوز متجاوز می‌ماند، و نقص‌های فیزیکی‌شان مانع اخلاق‌مدار بودنشان می‌شود. پس این به فردی اخلاق‌مدار و عاقل اختیار مطلق می‌دهد که جلوی متجاوز یا سادیست را بگیرد و هر جایی برای پذیرش اخلاقی ذهنی را کاملاً از میان می‌برد. آن‌ها ماشین‌هایی معیوب‌اند که آسیب می‌رسانند و باید متوقف شوند. همچنین تضمین می‌کند که تا وقتی هستی هست، متجاوزان، سادیست‌ها و جنایتکاران خواهند بود. افرادی با نقص‌های فیزیکی تولیدمثل می‌کنند و افرادی با نقص بیشتر و بیشتر پدید می‌آورند، آن‌ها غیراخلاقی، بی‌اخلاق و غیرعقلانی خواهند بود.

راهبرد: بسیاری از جنبش‌ها رویکرد گرواندن را دنبال می‌کنند. اما جبرگرایی عصب‌زیستی به عنوان واقعیتی مطلق این رویکرد را کاملاً نابود می‌کند. بسیاری از جنبش‌ها، مانند حقوق حیوانات، می‌گویند "وگان شو"، ضدزادآوری می‌گوید "ضدزادآور شو". اما حقیقت این است که نمی‌توانید کسی را که از نظر فیزیکی معیوب است قانع کنید از اخلاقی پیروی کند. چگونه کسی که از نظر فیزیکی نژادپرست یا از نظر فیزیکی سادیست است می‌تواند به خیر حیوانات فکر کند؟ راهبرد جنبش باید فقط رسیدن به افراد همدل و عاقل باشد. توانمندان. آن‌ها بی‌درنگ برای آرمان کار خواهند کرد.

راهبرد علمی: برای تعیین فناورانهٔ روشی که مؤثرترین نتیجه را در پدید آوردن آرامش کامل از راه انقراض دارد، جنبش باید از تولیدمثل انسان، به‌ویژه انقراض‌گرایان، پشتیبانی کند. بله، فرزندآوری بد است. اما گاهی وقتی موجودات حس‌مند دیگر را در نظر می‌گیریم، به شری ضروری بدل می‌شود. همچنین، همان‌طور که پیش‌تر گفتیم، فرزند یک آدم همدل به احتمال بیشتری همدل است. فرزند یک آدم عاقل به احتمال بیشتری عاقل است. به هر حال، نقص‌های فیزیکی قربانیان بیشتری به دنیا می‌آورند و جای شما را می‌گیرند، پس چه چیزی بهتر است؟ تولیدمثل یک آدم همدل یا یک خودشیفته؟ جهانی را تصور کنید که فرقه‌گرایانی مانند اسلام‌گرایان آن را تسخیر کرده‌اند، جهانی که خزانهٔ ژنی خودشیفته‌ها بر آن چیره است. این ویران‌شهری بی‌بازگشت خواهد بود که جامعه و جنبش را قرن‌ها به عقب می‌برد.

نکته

این‌جا دربارهٔ جبرگرایی کوانتومی بحث نمی‌کنم. اما عصب‌زیست‌شناسی، که زیر مکانیک کلاسیک قرار می‌گیرد، کاملاً جبری است. عصب‌زیست‌شناسی زیر مکانیک کلاسیک کار می‌کند.

پیامدگرایی

این اصلی بسیار بسیار ساده است که با یک ضرب‌المثل بهتر از همه تعریف می‌شود: "هدف وسیله را توجیه می‌کند."

برای نمونه، اگر فکر می‌کنید قتل نادرست است، نخستین کاری که باید بکنید کشتن هیتلر است! در عمل، فقط نتیجه‌ها می‌توانند چیزی را بهتر کنند. چون جهانی با حیات همچنان هیتلرها به شما خواهد داد و نتیجهٔ انقراض جهانی می‌تواند به این پایان دهد.

برای نمونه جنبش ضدبرده‌داری را در نظر بگیرید! آیا جنگ می‌خواستند؟ آیا شورش می‌خواستند؟ به هیچ وجه! اگر برتری‌طلبان سفیدپوست فقط بردگانشان را رها می‌کردند، بسیار خوشحال می‌شدند. اما این هرگز رخ نمی‌دهد. پس فکر کردند: چه چیزی بدتر است، ادامهٔ برده‌داری تا بی‌نهایت یا جنگی پایانی برای پایان دادن به آن؟ این پیامدگرایی است. اگر کسی فکر کند قتل نادرست است و هدف وسیله را توجیه نمی‌کند، ابلهانه می‌کوشد قاتلان را قانع کند سلاحشان را زمین بگذارند، که به هیچ جا نمی‌رسد. آدم‌های بیشتری آسیب خواهند دید.

پس این چگونه به هواداری از انقراض مربوط می‌شود؟

این اصل برای اخلاق و راهبرد جنبش، اکنون و در آینده، کاملاً ضروری است. چون انقراض‌گرایی وسیلهٔ پایانی همیشگی برای رنج است. صلح همیشگی برای همه.

ما پیش‌تر به کار بستن پیامدگرایی را در پیشروی به سوی انقراض کامل آغاز کرده‌ایم.

نخستین نمونه خود انقراض‌گرایی است. آیا مرگ دردناک بد است؟ بله. پس چه چیزی بدتر است؟ پایان دادن همیشگی به مرگ‌ها با انقراض یا ادامهٔ نظامی که حیات بیشتری را به مردن وامی‌دارد؟ آشکارا هر آدم عاقلی مرگ و رنج کمتر را برمی‌گزیند. آیا تولیدمثل بد است؟ بله، زندگی و رنج را بر یک موجود حس‌مند تحمیل می‌کند. اما 99.9 درصد رنج‌برندگان حیوانات‌اند و رنج شاید در کیهان گسترده باشد. پس به خیر بزرگ‌تر فکر می‌کنیم و تولیدمثل انسان را ادامه می‌دهیم تا تعیین کنیم گسترده‌ترین و کامل‌ترین انقراض ممکن چیست. ما انقراض‌گرایان برای آرمان‌های کوچک نمی‌جنگیم، چون اختصاص دادن نیروی انسانی اندکی که داریم به جنبشی مانند LGBT یا فمینیسم حتی رنج را کم نمی‌کند و به‌راستی به جلوگیری از زندگی بد همه نمی‌رسد. جنبش برای انقراض کامل ممکن است به همین دلیل به تأخیر بیفتد یا به خطر بیفتد. پس بر حل رنج همه (تنها مشکل همهٔ زمان‌ها) تمرکز می‌کنیم.

پیامدگرایی برای راهبرد و اخلاق حیاتی است. همیشه غیراخلاقی است که به جای رنج کمتر، رنج بیشتری را روا بداریم (یا پدید آوریم).

ماده‌گرایی و جبرگرایی به نتیجهٔ راهبردی گریزناپذیر پیامدگرایی می‌انجامند. رنج در این طبیعت تعبیه شده است. و چیزها همان‌طور رخ می‌دهند که قانون‌های طبیعی حکم می‌کنند. و آن‌ها حکم می‌کنند که رنج گریزناپذیر است. زندگی آن‌قدر سیاه و سفید نیست، نمی‌گذارد همه چیز را صلح‌آمیز به دست آورید. نژادپرستان سر راه ما خواهند ایستاد. طبیعت همچنان رنج را گسترش خواهد داد. پس گاهی ناچار خواهیم بود شر کمتر را برگزینیم، مانند نمونهٔ بالا. و باید بی‌درنگ چنین کنیم. پایانی همیشگی برای رنج پیامدی بسیار مهم‌تر از هر وسیله‌ای است.

روش علمی

چرا روش علمی مهم است؟ چون این جنبش نهایی است، می‌خواهیم صلح نهایی را برسانیم. و نمی‌توانیم در آن شتاب‌زده باشیم.

همان‌طور که در مانیفستمان آمده، می‌خواهیم گسترده‌ترین و کامل‌ترین انقراض ممکن را رقم بزنیم. پس چه کسی آن را تعیین می‌کند؟ پاسخ علم است. هیچ‌کس نمی‌تواند بی شواهد علمی وارد شود و بگوید "این گسترده‌ترین و کامل‌ترین انقراض ممکن است". اگر چنین شود، کل جنبش شکست می‌خورد. یک فیزیک‌دان پوچ‌گرای تصادفی می‌آید و می‌گوید "فقط بمب اتمی بزنید، تنها چیز ممکن همین است." این فیزیک‌دان آشکارا غیرعقلانی است و به رنج می‌انجامد، نه به انقراض. فقط روش علمی می‌تواند آن را تعیین کند. نه یک فرد یا یک گروه.

میان روش علمی و یک دانشمند تفاوتی روشن هست. نباید به دانشمندان (افراد) اعتماد کرد. بنابراین ما فقط انتشارات عقلانی و نتیجه‌های آزمایشی را می‌پذیریم که به‌طور مستقل بازتولید و به‌طور تجربی راستی‌آزمایی شده‌اند.

اگر ابلهی مجله‌ای برای هومیوپاتی یا دیگر شبه‌علم‌ها راه بیندازد، آن‌ها را نمی‌توان در نظر گرفت. و انقراض‌گرایی همیشه شبه‌علم‌هایی مانند هومیوپاتی، طالع‌بینی، کف‌بینی، ضدواکسن، فراروان‌شناسی و مانند آن‌ها را رد می‌کند. راه دادن این چیزها به هر جنبشی مانند سوزاندن کل جنبش با دست خودتان است. این‌ها قاتل عقلانیت‌اند. هرگز به راه‌حلی نمی‌رسند. پس انقراض‌گرایان فقط به روش علمی عقلانی اعتماد می‌کنند.

خداناباوری

خداباوری بر باور بنا شده، بر ایمان بنا شده است. دین و ایدهٔ خدا کاملاً ساختهٔ انسان‌اند. انسان‌ها گرایش دارند وقتی پاسخی ندارند جاهای خالی را با جادو پر کنند. گرایش داریم وقتی هیچ شاهدی نداریم پاسخ جا بیندازیم. این‌گونه دین‌ها را ساختیم.

منطق 4000 سال پیش از میلاد: "او نفرین شده، حتماً چون به الههٔ ماه بی‌احترامی کرده."

علم امروز: "این بیماری‌ای است که ریزجاندارانی پدید می‌آورند که با چشم غیرمسلح دیده نمی‌شوند."

همهٔ پدیده‌های واقعیت مادی توضیحی علمی دارند. و کسانی که جاهای خالی را با خدا، بتمن یا گودزیلا پر می‌کنند هیچ جایی در جنبشی مبتنی بر عقلانیت ندارند.

بسیار پوچ است که حتی در قرن بیست‌ویکم مردم موجودی را می‌پرستند که گویا هستی را آفریده، به کودکانشان تجاوز می‌کند، حیوانات را آزار می‌دهد، حیوانات را آفریده تا یکدیگر را بکشند، بلایای طبیعی را آفریده و مانند آن. هیچ نقشهٔ بزرگی به ذهنم نمی‌رسد که بتواند تجاوز به کودکان و شکنجهٔ توله‌سگ‌ها و شکار را توجیه کند.

همچنین این پرسش هست: چه جور موجود ابرهوشمندی هستی‌ای بی‌معنا و غیرعقلانی آفرید که بی‌دلیل فقط رنج پدید می‌آورد؟

بسیار پوچ است که 90% مردم در قرن بیست‌ویکم خداباورند. و بدتر این‌که به خدایی نیکوکار و خوب باور دارند. در خداباوری باورهای پوچ‌تری هم هست، مانند دعا. 99% خداباوران به خدایی نیکوکار باور دارند که به آن‌ها ماشین یا خانه می‌بخشد، اما بردگان جنسی و گرسنگان را آزاد نمی‌کند.

هیچ خدایی برای نجات رنج‌برندگان نمی‌آید، این وظیفهٔ اخلاقی ما به عنوان انسان است، چون ما تنها کسانی هستیم که وجود داریم و تنها کسانی که می‌توانیم به موجودات دیگر کمک کنیم.

پس چرا این برای انقراض‌گرایی مهم است؟

انقراض‌گرایان باید عقل‌گرایانی باشند که به شواهد مادی باور دارند و به معجزه امید نمی‌بندند. یک انقراض‌گرای خداباور می‌تواند فردا ادعا کند "خدایم به خوابم آمد و گفت انقراض دیگر آرمانی ارزشمند نیست" و اصول ما دقیقاً برای جلوگیری از پیوستن چنین ابلهانی به جنبش طراحی شده‌اند.
